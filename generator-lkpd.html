function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu('⚡ Menu Prompt')
    .addItem('🚀 Generate Prompt LKPD', 'generatePrompt')
    .addItem('📋 Copy Prompt ke Clipboard', 'copyPrompt')

/**
 * Fungsi untuk menyalin isi sel A1 Sheet "Prompt" langsung ke Clipboard komputer user
 * menggunakan modal dialog HTML dan Clipboard API browser.
 */
function copyPrompt() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const promptSheet = ss.getSheetByName('Prompt');

  if (!promptSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error: Sheet "Prompt" tidak ditemukan!');
    return;
  }

  const promptText = promptSheet.getRange('A1').getValue();

  if (!promptText || promptText.toString().trim() === '') {
    SpreadsheetApp.getUi().alert(
      '⚠️ Prompt Masih Kosong!',
      'Silakan jalankan "🚀 Generate Prompt LKPD" terlebih dahulu dari menu.',
      SpreadsheetApp.getUi().ButtonSet.OK
    );
    return;
  }

  // Membuat halaman HTML interaktif untuk mengeksekusi perintah copy ke clipboard
  const htmlContent = `
    
    
      
        
        
      
      
        

          
📋

          
Proses Menyalin Prompt...

          
Mohon tunggu sebentar...

          
          
        


        
      
    
  `;

  const htmlOutput = HtmlService.createHtmlOutput(htmlContent)
    .setWidth(420)
    .setHeight(230);

  SpreadsheetApp.getUi().showModalDialog(htmlOutput, '📋 Copy Prompt LKPD');
}

Tips Debugging & Anti-Error

1. Kenapa menu '⚡ Menu Prompt' tidak muncul?

Pastikan Anda telah merefresh halaman Google Sheets. Jika belum muncul, buka Apps Script dan jalankan fungsi onOpen() secara manual sekali.

2. Hasil Sertifikat Gambar HD 3:4:

AI diinstruksikan menggunakan library html2canvas dengan devicePixelRatio 2 untuk menghasilkan sertifikat PNG tajam berukuran 900×1200 px.

3. Rekomendasi Model AI:

Gunakan Claude 3.5 Sonnet atau ChatGPT-4o karena memiliki kemampuan merender single-file index.html tanpa terpotong untuk prompt panjang.
