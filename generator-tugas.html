/**
 * Fungsi utama untuk membaca data dari Sheet "Input",

function generatePrompt() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const inputSheet = ss.getSheetByName('Input');
  const promptSheet = ss.getSheetByName('Prompt');

  // Validasi keberadaan sheet
  if (!inputSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error: Sheet "Input" tidak ditemukan. Pastikan nama sheet sesuai!');
    return;
  }
  if (!promptSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error: Sheet "Prompt" tidak ditemukan. Pastikan nama sheet sesuai!');
    return;
  }

  // Mengambil nilai input dari range B1:B21
  const values = inputSheet.getRange('B1:B21').getValues();

  // Helper function untuk mengambil nilai sel dengan fallback jika kosong
  const getValue = (index, fallback) => {
    const val = values[index] ? values[index][0] : '';
    return (val !== null && val !== undefined && val.toString().trim() !== '') 
      ? val.toString().trim() 
      : fallback;
  };

  // Ekstraksi variabel dari sheet Input
  const mapel          = getValue(0, '[MAPEL]');
  const materi         = getValue(1, '[NAMA MATERI]');
  const tujuan1        = getValue(2, '[TUJUAN 1]');
  const tujuan2        = getValue(3, '[TUJUAN 2]');
  const tujuan3        = getValue(4, '[TUJUAN 3]');
  const tujuan4        = getValue(5, '[TUJUAN 4]');
  const tujuan5        = getValue(6, '[TUJUAN 5]');
  const subtopikMudah  = getValue(7, '[SUBTOPIK MUDAH]');
  const subtopikNormal = getValue(8, '[SUBTOPIK NORMAL]');
  const subtopikSulit  = getValue(9, '[SUBTOPIK SULIT]');
  const karakterMudah  = getValue(10, '[NAMA KARAKTER MUDAH]');
  const karakterNormal = getValue(11, '[NAMA KARAKTER NORMAL]');
  const karakterSulit  = getValue(12, '[NAMA KARAKTER SULIT]');
  const warna1         = getValue(13, '[WARNA 1]');
  const warna2         = getValue(14, '[WARNA 2]');
  const warna3         = getValue(15, '[WARNA 3]');
  const ikon1          = getValue(16, '[IKON 1]');
  const ikon2          = getValue(17, '[IKON 2]');
  const ikon3          = getValue(18, '[IKON 3]');
  const ikon4          = getValue(19, '[IKON 4]');

10. Semua tag HTML ditutup benar.

/**
 * Fungsi untuk menyalin isi sel A1 Sheet "Prompt" langsung ke Clipboard komputer user
 * menggunakan modal dialog HTML interaktif dan Navigator Clipboard API.
 */
function copyPrompt() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const promptSheet = ss.getSheetByName('Prompt');

  if (!promptSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error: Sheet "Prompt" tidak ditemukan!');
    return;
  }

  const promptText = promptSheet.getRange('A1').getValue();

  if (!promptText || promptText.toString().trim() === '') {
    SpreadsheetApp.getUi().alert(
      '⚠️ Prompt Masih Kosong!',
      'Silakan jalankan "🚀 Generate Prompt Tugas 3 Level" terlebih dahulu dari menu.',
      SpreadsheetApp.getUi().ButtonSet.OK
    );
    return;
  }

  // Membuat halaman HTML interaktif untuk mengeksekusi perintah copy ke clipboard
  const htmlContent = `
    
    
      
        
        
      
      
        

          
📋

          
Proses Menyalin Prompt 3 Level...

          
Mohon tunggu sebentar...

          
          
        


        
      
    
  `;

  const htmlOutput = HtmlService.createHtmlOutput(htmlContent)
    .setWidth(430)
    .setHeight(240);

  SpreadsheetApp.getUi().showModalDialog(htmlOutput, '📋 Copy Prompt Tugas 3 Level');
}
